/** Solo para una instalación nueva desde cero. */
function instalar() {
  const p = props();
  if (p.getProperty('SS')) {
    Logger.log('Ya estaba instalado. Ejecuta "configurar" y luego "crearEnlaces".');
    return;
  }
  const carpeta = DriveApp.createFolder(NOMBRE + ' (privado)');
  const archivos = carpeta.createFolder('Archivos');
  const ss = SpreadsheetApp.create(NOMBRE + ' - datos');
  DriveApp.getFileById(ss.getId()).moveTo(carpeta);
  ss.getSheets()[0].setName('Eventos');
  const cal = CalendarApp.createCalendar(NOMBRE_CALENDARIO, { timeZone: ZONA });
  p.setProperties({ SS: ss.getId(), CARPETA: archivos.getId(), CAL: cal.getId() });
  configurar();
  crearEnlaces();
}

/**
 * Prepara la hoja para las funciones nuevas y activa el respaldo de cada domingo.
 * Se puede ejecutar las veces que quieras sin dañar nada.
 */
function configurar() {
  const ss = SpreadsheetApp.openById(props().getProperty('SS'));
  Object.keys(COLUMNAS).forEach(function (nombre) {
    let h = ss.getSheetByName(nombre);
    if (!h) {
      h = ss.insertSheet(nombre);
      h.getRange('A:Z').setNumberFormat('@');
      h.appendRow(COLUMNAS[nombre]);
      h.setFrozenRows(1);
      return;
    }
    h.getRange('A:Z').setNumberFormat('@');
    const ultima = Math.max(1, h.getLastColumn());
    const cab = h.getRange(1, 1, 1, ultima).getValues()[0].map(String);
    COLUMNAS[nombre].forEach(function (c) {
      if (cab.indexOf(c) < 0) { cab.push(c); h.getRange(1, cab.length).setValue(c); }
    });
  });

  // Respaldo cada domingo de madrugada y resumen por correo cada domingo en la mañana
  ScriptApp.getProjectTriggers().forEach(function (t) {
    const f = t.getHandlerFunction();
    if (f === 'respaldar' || f === 'resumenSemanal') ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger('respaldar').timeBased().onWeekDay(ScriptApp.WeekDay.SUNDAY).atHour(3).create();
  ScriptApp.newTrigger('resumenSemanal').timeBased().onWeekDay(ScriptApp.WeekDay.SUNDAY).atHour(8).create();
  respaldar();
  Logger.log('Listo. Respaldo y resumen de cada domingo activados. Primer respaldo hecho.');
}

/**
 * Crea una clave para cada persona de la lista y muestra su enlace.
 * Las claves que ya existen no cambian.
 */
function crearEnlaces() {
  const p = props();
  if (!p.getProperty('SS')) throw new Error('Primero ejecuta "instalar".');
  const claves = JSON.parse(p.getProperty('CLAVES') || '{}');
  USUARIOS.forEach(function (u) {
    if (!claves[u[0]]) claves[u[0]] = Utilities.getUuid().replace(/-/g, '').slice(0, 16);
  });
  p.setProperty('CLAVES', JSON.stringify(claves));
  p.deleteProperty('CLAVE');
  Logger.log('Enlaces (manda a cada persona SOLO el suyo):');
  USUARIOS.forEach(function (u) {
    Logger.log(u[0] + (u[1] === true ? ' (admin)' : u[1] === 'mama' ? ' (vista para mamá)' : '') + ':  ' + DIRECCION_APP + '?k=' + claves[u[0]]);
  });
}

/** Si un enlace se filtra: crea claves nuevas para todos y las viejas dejan de funcionar. */
function cambiarTodasLasClaves() {
  props().deleteProperty('CLAVES');
  crearEnlaces();
}

/** Copia la hoja de datos a la carpeta "Respaldos". Se ejecuta sola cada domingo. */
function respaldar() {
  const p = props();
  const privada = DriveApp.getFolderById(p.getProperty('CARPETA')).getParents().next();
  let carpeta;
  try { carpeta = DriveApp.getFolderById(p.getProperty('RESPALDOS')); } catch (x) { carpeta = null; }
  if (!carpeta) {
    carpeta = privada.createFolder('Respaldos');
    p.setProperty('RESPALDOS', carpeta.getId());
  }
  const fecha = Utilities.formatDate(new Date(), ZONA, 'yyyy-MM-dd');
  DriveApp.getFileById(p.getProperty('SS')).makeCopy('Respaldo ' + fecha, carpeta);
  p.setProperty('ULTIMO_RESPALDO', new Date().toISOString());

  // Guardar solo los últimos respaldos
  const lista = [];
  const it = carpeta.getFiles();
  while (it.hasNext()) lista.push(it.next());
  lista.sort(function (a, b) { return b.getDateCreated() - a.getDateCreated(); });
  lista.slice(RESPALDOS_A_GUARDAR).forEach(function (f) { f.setTrashed(true); });

  vaciarPapeleraVieja();
}

function usuario(k) {
  if (!k) return null;
  const claves = JSON.parse(props().getProperty('CLAVES') || '{}');
  for (let i = 0; i < USUARIOS.length; i++) {
    if (claves[USUARIOS[i][0]] === k) {
      return { nombre: USUARIOS[i][0], admin: USUARIOS[i][1] === true, mama: USUARIOS[i][1] === 'mama' };
    }
  }
  return null;
}

function hoja(nombre) {
  return SpreadsheetApp.openById(props().getProperty('SS')).getSheetByName(nombre);
}

/** Lee una hoja completa como lista de objetos según su encabezado. */
function tabla(nombre) {
  const h = hoja(nombre);
  if (!h) return { h: null, cab: [], filas: [] };
  const v = h.getDataRange().getValues();
  const cab = v.shift().map(String);
  return {
    h: h, cab: cab,
    filas: v.map(function (r, i) {
      const o = { _fila: i + 2 };
      cab.forEach(function (c, j) { o[c] = r[j] == null ? '' : String(r[j]); });
      return o;
    })
  };
}

/** Escribe un objeto en la fila indicada (o al final si fila < 1), según el encabezado. */
function escribir(t, fila, obj) {
  const row = t.cab.map(function (c) { return obj[c] == null ? '' : String(obj[c]); });
  if (fila > 0) t.h.getRange(fila, 1, 1, row.length).setValues([row]);
  else t.h.appendRow(row);
}

function filaDe(nombre, id) {
  const v = hoja(nombre).getRange('A:A').getValues();
  for (let i = 1; i < v.length; i++) if (String(v[i][0]) === id) return i + 1;
  return -1;
}

/** Borra las filas donde la columna llamada `campo` vale `valor`. Devuelve los objetos borrados. */
function borrarDonde(nombre, campo, valor) {
  const t = tabla(nombre);
  const borradas = t.filas.filter(function (o) { return o[campo] === valor; });
  borradas.slice().sort(function (a, b) { return b._fila - a._fila; })
    .forEach(function (o) { t.h.deleteRow(o._fila); });
  return borradas;
}

/** Manda a la papelera de Drive los archivos (se recuperan 30 días desde Drive). */
function aPapeleraDrive(archivos) {
  archivos.forEach(function (a) {
    try { DriveApp.getFileById(a.driveId).setTrashed(true); } catch (x) {}
  });
}

function json(o) {
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}

function fechaCorta(iso) {
  try { return Utilities.formatDate(new Date(iso), ZONA, 'dd/MM/yyyy'); } catch (x) { return ''; }
}

function dinero(n) { return '$' + (Number(n) || 0).toFixed(2); }

function registrar(quien, texto) {
  const h = hoja('Historial');
  if (h) h.appendRow([new Date().toISOString(), quien, texto]);
}

/** Crea o actualiza el evento en el calendario compartido. Devuelve su id. */
function sincronizarCalendario(ev) {
  const cal = CalendarApp.getCalendarById(props().getProperty('CAL'));
  const inicio = new Date(ev.inicio);
  const fin = new Date(inicio.getTime() + (Number(ev.duracionMin) || 120) * 60000);
  const desc = [ev.novedades, ev.siguientePaso ? 'Siguiente paso: ' + ev.siguientePaso : '']
    .filter(String).join('\n\n') + '\n\nMás detalles en el Cuaderno familia.';
  let ce = null;
  try { ce = ev.calId ? cal.getEventById(ev.calId) : null; } catch (x) { ce = null; }
  if (ce) {
    ce.setTitle(ev.titulo);
    ce.setTime(inicio, fin);
    ce.setDescription(desc);
    return ev.calId;
  }
  ce = cal.createEvent(ev.titulo, inicio, fin, { description: desc });
  ce.removeAllReminders();
  ce.addPopupReminder(24 * 60);
  ce.addPopupReminder(120);
  return ce.getId();
}

function quitarDelCalendario(calId) {
  try { CalendarApp.getCalendarById(props().getProperty('CAL')).getEventById(calId).deleteEvent(); } catch (x) {}
}

/** Borra para siempre lo que lleva más de 30 días en la papelera. */
function vaciarPapeleraVieja() {
  const limite = Date.now() - DIAS_PAPELERA * 864e5;
  tabla('Eventos').filas.forEach(function (e) {
    if (e.borrado && new Date(e.borrado).getTime() < limite) {
      borrarDonde('Eventos', 'id', e.id);
      borrarDonde('Pagos', 'eventoId', e.id);
      aPapeleraDrive(borrarDonde('Archivos', 'eventoId', e.id));
      registrar('Sistema', 'Vació de la papelera «' + e.titulo + '» (' + fechaCorta(e.inicio) + ')');
    }
  });
}

function sinCampos(o, quitar) {
  const r = {};
  Object.keys(o).forEach(function (k) { if (k[0] !== '_' && quitar.indexOf(k) < 0) r[k] = o[k]; });
  return r;
}

function doGet(e) {
  const q = (e && e.parameter) || {};
  const u = usuario(q.k);
  if (!u) return json({ error: 'Clave incorrecta' });
  const p = props();

  if (q.accion === 'foto') {
    const m = tabla('Mensajes').filas.find(function (x) { return x.id === q.id && x.fotoId; });
    if (!m) return json({ error: 'Esa foto ya no existe' });
    const blob = DriveApp.getFileById(m.fotoId).getBlob();
    return json({ tipo: m.fotoTipo, base64: Utilities.base64Encode(blob.getBytes()) });
  }
  const mensajes = tabla('Mensajes').filas.slice(-100).reverse().map(function (m) {
    return { id: m.id, fecha: m.fecha, quien: m.quien, texto: m.texto, foto: m.fotoId ? true : false };
  });

  const eventos = tabla('Eventos').filas;

  if (u.mama) {
    // Vista tranquila: solo citas próximas y recientes, sin detalles médicos ni pagos.
    const desde = Date.now() - 14 * 864e5;
    return json({
      yo: u,
      mensajes: mensajes,
      eventos: eventos.filter(function (x) { return !x.borrado && x.tipo !== 'nota' && new Date(x.inicio).getTime() > desde; })
        .map(function (x) { return { id: x.id, tipo: x.tipo, inicio: x.inicio, titulo: x.titulo, acompanan: x.acompanan }; })
    });
  }

  const activo = {};
  eventos.forEach(function (x) { if (!x.borrado) activo[x.id] = true; });
  const pagos = tabla('Pagos').filas.filter(function (x) { return activo[x.eventoId]; });
  const archivos = tabla('Archivos').filas;
  const visible = {};
  pagos.forEach(function (x) { if (u.admin || x.quienPago === u.nombre) visible[x.id] = true; });
  // Los archivos de la sesión los ven todos; los comprobantes, solo quien pagó y los admin.
  const puedeVer = function (a) { return u.admin || (activo[a.eventoId] && (!a.pagoId || visible[a.pagoId])); };

  if (q.accion === 'archivo') {
    const a = archivos.find(function (x) { return x.id === q.id; });
    if (!a || !puedeVer(a)) return json({ error: 'Ese archivo ya no existe' });
    const blob = DriveApp.getFileById(a.driveId).getBlob();
    return json({ nombre: a.nombre, tipo: a.tipo, base64: Utilities.base64Encode(blob.getBytes()) });
  }

  const totales = {};
  pagos.forEach(function (x) { totales[x.eventoId] = (totales[x.eventoId] || 0) + (Number(x.monto) || 0); });

  const res = {
    yo: u,
    ficha: JSON.parse(p.getProperty('FICHA') || '{}'),
    eventos: eventos.filter(function (x) { return !x.borrado; })
      .map(function (x) { return sinCampos(x, ['calId', 'borrado', 'borradoPor']); }),
    pagos: pagos.filter(function (x) { return visible[x.id]; }).map(function (x) { return sinCampos(x, []); }),
    archivos: archivos.filter(function (a) { return activo[a.eventoId] && puedeVer(a); })
      .map(function (a) { return sinCampos(a, ['driveId']); }),
    totales: totales,
    mensajes: mensajes
  };
  if (u.admin) {
    res.papelera = eventos.filter(function (x) { return x.borrado; }).map(function (x) {
      return { id: x.id, tipo: x.tipo, titulo: x.titulo, inicio: x.inicio, borrado: x.borrado, borradoPor: x.borradoPor };
    });
    res.historial = tabla('Historial').filas.slice(-300).reverse().map(function (x) { return sinCampos(x, []); });
    res.ultimoRespaldo = p.getProperty('ULTIMO_RESPALDO') || '';
  }
  return json(res);
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    const d = JSON.parse(e.postData.contents);
    const u = usuario(d.k);
    if (!u) return json({ error: 'Clave incorrecta' });
    if (d.accion === 'mensaje') return json(nuevoMensaje(d, u));
    if (d.accion === 'borrarMensaje') return json(borrarMensaje(d.id, u));
    if (u.mama) return json({ error: 'Esta acción no está disponible.' });
    if (d.accion === 'acompanar') return json(acompanar(d, u));
    if (d.accion === 'guardar') return json(guardar(d, u));
    if (d.accion === 'borrar') return json(borrar(d.id, u));
    if (d.accion === 'restaurar') return json(restaurar(d.id, u));
    if (d.accion === 'pendiente') return json(pendiente(d, u));
    if (d.accion === 'ficha') return json(guardarFicha(d.ficha, u));
    return json({ error: 'Acción desconocida' });
  } catch (err) {
    return json({ error: String(err) });
  } finally {
    lock.releaseLock();
  }
}

function guardar(d, u) {
  const ev = d.evento;
  const ahora = new Date().toISOString();
  const tE = tabla('Eventos');
  const id = ev.id || Utilities.getUuid();
  const previa = tE.filas.find(function (x) { return x.id === id; }) || null;
  if (previa && previa.borrado) return { error: 'Ese registro está en la papelera. Un administrador puede restaurarlo.' };

  const sint = ev.sintomas && Object.keys(ev.sintomas).some(function (k) { return ev.sintomas[k]; });
  const nuevo = Object.assign({}, previa || {}, {
    id: id,
    tipo: ev.tipo,
    inicio: ev.inicio,
    duracionMin: String(Number(ev.duracionMin) || 120),
    titulo: ev.titulo,
    novedades: ev.novedades || '',
    siguientePaso: ev.siguientePaso || '',
    quien: previa ? previa.quien : u.nombre,
    creado: previa ? previa.creado : ahora,
    sintomas: sint ? JSON.stringify(ev.sintomas) : '',
    ciclo: ev.ciclo ? String(ev.ciclo) : ''
  });
  if (previa && previa.siguientePaso !== nuevo.siguientePaso) nuevo.pasoHecho = '';
  nuevo.calId = sincronizarCalendario(nuevo);
  escribir(tE, previa ? previa._fila : -1, nuevo);

  const log = [(previa ? 'Editó' : 'Agregó') + ' «' + ev.titulo + '» (' + fechaCorta(ev.inicio) + ')'];

  // ---- Pagos ---- Un admin maneja todos los de la sesión; los demás, solo los suyos.
  const hp = hoja('Pagos');
  const existentes = tabla('Pagos').filas.filter(function (x) { return x.eventoId === id; });
  const mios = {}, deOtros = {}, porId = {};
  existentes.forEach(function (x) {
    porId[x.id] = x;
    if (u.admin || x.quienPago === u.nombre) mios[x.id] = true; else deOtros[x.id] = true;
  });
  const entrantes = (d.pagos || []).filter(function (x) { return !deOtros[x.id]; });
  const vienen = {};
  entrantes.forEach(function (x) { if (x.id) vienen[x.id] = true; });

  Object.keys(mios).forEach(function (pid) {
    if (!vienen[pid]) {
      const viejo = porId[pid];
      borrarDonde('Pagos', 'id', pid);
      aPapeleraDrive(borrarDonde('Archivos', 'pagoId', pid));
      delete mios[pid];
      log.push('quitó un pago de ' + dinero(viejo.monto) + (viejo.quienPago ? ' (' + viejo.quienPago + ')' : ''));
    }
  });
  entrantes.forEach(function (x) {
    const pid = x.id || Utilities.getUuid();
    const quienPago = u.admin ? (x.quienPago || '') : u.nombre;
    const r = [pid, id, String(x.monto || ''), x.forma || '', x.detalle || '', quienPago, ahora];
    const viejo = porId[pid];
    const f = viejo ? filaDe('Pagos', pid) : -1;
    if (f > 0) {
      hp.getRange(f, 1, 1, r.length).setValues([r]);
      if (Number(viejo.monto) !== Number(x.monto) || viejo.quienPago !== quienPago || viejo.forma !== x.forma) {
        log.push('cambió un pago de ' + dinero(viejo.monto) + ' (' + (viejo.quienPago || 'sin nombre') + ') a ' +
          dinero(x.monto) + ' (' + (quienPago || 'sin nombre') + ', ' + (x.forma || '') + ')');
      }
    } else {
      hp.appendRow(r);
      log.push('agregó un pago de ' + dinero(x.monto) + (quienPago ? ' (' + quienPago + ', ' + (x.forma || '') + ')' : ''));
    }
    mios[pid] = true;
  });

  // ---- Archivos nuevos ----
  const carpeta = DriveApp.getFolderById(props().getProperty('CARPETA'));
  const ha = hoja('Archivos');
  let subidos = 0;
  (d.archivos || []).forEach(function (a) {
    if (a.pagoId && !mios[a.pagoId]) return; // comprobante de un pago ajeno: no se guarda
    const blob = Utilities.newBlob(Utilities.base64Decode(a.base64), a.tipo, a.nombre);
    const f = carpeta.createFile(blob);
    ha.appendRow([Utilities.getUuid(), id, a.nombre, a.tipo, f.getId(), ahora, a.pagoId || '']);
    subidos++;
  });
  if (subidos) log.push('subió ' + subidos + (subidos === 1 ? ' archivo' : ' archivos'));

  // ---- Archivos que alguien quitó ---- (admin: cualquiera; los demás: comprobantes de sus pagos)
  if ((d.quitarArchivos || []).length) {
    const todos = tabla('Archivos').filas;
    d.quitarArchivos.forEach(function (aid) {
      const a = todos.find(function (x) { return x.id === aid; });
      if (!a) return;
      if (u.admin || (a.pagoId && mios[a.pagoId])) {
        aPapeleraDrive(borrarDonde('Archivos', 'id', aid));
        log.push('quitó el archivo «' + a.nombre + '»');
      }
    });
  }

  registrar(u.nombre, log.join('; '));
  return { ok: true, id: id };
}

function borrar(id, u) {
  if (!u.admin) return { error: 'Solo un administrador puede borrar registros completos.' };
  const t = tabla('Eventos');
  const ev = t.filas.find(function (x) { return x.id === id; });
  if (!ev) return { ok: true };
  quitarDelCalendario(ev.calId);
  ev.calId = '';
  ev.borrado = new Date().toISOString();
  ev.borradoPor = u.nombre;
  escribir(t, ev._fila, ev);
  registrar(u.nombre, 'Mandó a la papelera «' + ev.titulo + '» (' + fechaCorta(ev.inicio) + ')');
  return { ok: true };
}

function restaurar(id, u) {
  if (!u.admin) return { error: 'Solo un administrador puede restaurar.' };
  const t = tabla('Eventos');
  const ev = t.filas.find(function (x) { return x.id === id; });
  if (!ev) return { error: 'Ya no está en la papelera.' };
  ev.borrado = '';
  ev.borradoPor = '';
  ev.calId = sincronizarCalendario(ev);
  escribir(t, ev._fila, ev);
  registrar(u.nombre, 'Restauró «' + ev.titulo + '» (' + fechaCorta(ev.inicio) + ')');
  return { ok: true };
}

function pendiente(d, u) {
  const t = tabla('Eventos');
  const ev = t.filas.find(function (x) { return x.id === d.id && !x.borrado; });
  if (!ev || !ev.siguientePaso) return { error: 'Ese pendiente ya no existe.' };
  ev.pasoHecho = d.hecho ? new Date().toISOString() + '|' + u.nombre : '';
  escribir(t, ev._fila, ev);
  registrar(u.nombre, (d.hecho ? 'Marcó como hecho: «' : 'Desmarcó: «') + ev.siguientePaso + '»');
  return { ok: true, pasoHecho: ev.pasoHecho };
}

function guardarFicha(f, u) {
  if (!u.admin) return { error: 'Solo un administrador puede cambiar la ficha.' };
  const limpia = {};
  CAMPOS_FICHA.forEach(function (c) { limpia[c] = String((f || {})[c] || '').slice(0, 3000); });
  props().setProperty('FICHA', JSON.stringify(limpia));
  registrar(u.nombre, 'Actualizó la ficha rápida');
  return { ok: true };
}

/* =========================================================
   Acompañantes
   ========================================================= */

function acompanar(d, u) {
  const t = tabla('Eventos');
  const ev = t.filas.find(function (x) { return x.id === d.id && !x.borrado; });
  if (!ev) return { error: 'Esa sesión ya no existe.' };
  let lista = ev.acompanan ? ev.acompanan.split('|') : [];
  lista = lista.filter(function (n) { return n !== u.nombre; });
  if (d.si) lista.push(u.nombre);
  ev.acompanan = lista.join('|');
  escribir(t, ev._fila, ev);
  registrar(u.nombre, (d.si ? 'Se anotó para acompañar' : 'Ya no acompaña') + ' en «' + ev.titulo + '» (' + fechaCorta(ev.inicio) + ')');
  return { ok: true, acompanan: ev.acompanan };
}

/* =========================================================
   Mensajes para mamá
   ========================================================= */

function nuevoMensaje(d, u) {
  const texto = String(d.texto || '').trim().slice(0, 2000);
  if (!texto && !d.foto) return { error: 'Escribe un mensaje o elige una foto.' };
  let fotoId = '', fotoTipo = '';
  if (d.foto && d.foto.base64) {
    const blob = Utilities.newBlob(Utilities.base64Decode(d.foto.base64), d.foto.tipo, d.foto.nombre || 'foto.jpg');
    fotoId = DriveApp.getFolderById(props().getProperty('CARPETA')).createFile(blob).getId();
    fotoTipo = d.foto.tipo;
  }
  const id = Utilities.getUuid();
  hoja('Mensajes').appendRow([id, new Date().toISOString(), u.nombre, texto, fotoId, fotoTipo]);
  registrar(u.nombre, 'Dejó un mensaje para mamá' + (fotoId ? ' con foto' : ''));
  return { ok: true, id: id };
}

function borrarMensaje(id, u) {
  const m = tabla('Mensajes').filas.find(function (x) { return x.id === id; });
  if (!m) return { ok: true };
  if (!u.admin && m.quien !== u.nombre) return { error: 'Solo puedes borrar tus propios mensajes.' };
  borrarDonde('Mensajes', 'id', id);
  if (m.fotoId) aPapeleraDrive([{ driveId: m.fotoId }]);
  registrar(u.nombre, 'Borró un mensaje de ' + m.quien);
  return { ok: true };
}

/* =========================================================
   Resumen de cada domingo por correo
   ========================================================= */

/** Para probar: ejecútala y te llega ahora mismo a los correos configurados. */
function enviarResumenAhora() { resumenSemanal(); }

function resumenSemanal() {
  const ahora = Date.now(), hace7 = ahora - 7 * 864e5, en14 = ahora + 14 * 864e5;
  const ficha = JSON.parse(props().getProperty('FICHA') || '{}');
  const eventos = tabla('Eventos').filas.filter(function (x) { return !x.borrado; });
  const pagos = tabla('Pagos').filas;
  const esMama = {};
  USUARIOS.forEach(function (x) { if (x[1] === 'mama') esMama[x[0]] = true; });
  const mensajes = tabla('Mensajes').filas.filter(function (m) { return new Date(m.fecha).getTime() > hace7 && !esMama[m.quien]; });
  const tipos = { inmuno: 'Inmunoterapia', quimio: 'Suero', cita: 'Cita médica', examen: 'Examen', nota: 'Nota' };
  const esc = function (t) { return String(t || '').replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); };
  const DIAS = ['domingo', 'lunes', 'martes', 'miércoles', 'jueves', 'viernes', 'sábado'];
  const MESES = ['enero', 'febrero', 'marzo', 'abril', 'mayo', 'junio', 'julio', 'agosto', 'septiembre', 'octubre', 'noviembre', 'diciembre'];
  const cuando = function (iso) {
    const d = new Date(iso);
    const n = function (f) { return Number(Utilities.formatDate(d, ZONA, f)); };
    return DIAS[n('u') % 7] + ' ' + n('d') + ' de ' + MESES[n('M') - 1] + ', ' + Utilities.formatDate(d, ZONA, 'HH:mm');
  };
  const t = function (iso) { return new Date(iso).getTime(); };

  const pasaron = eventos.filter(function (x) { return t(x.inicio) >= hace7 && t(x.inicio) <= ahora; })
    .sort(function (a, b) { return a.inicio < b.inicio ? -1 : 1; });
  const vienen = eventos.filter(function (x) { return t(x.inicio) > ahora && t(x.inicio) <= en14 && x.tipo !== 'nota'; })
    .sort(function (a, b) { return a.inicio < b.inicio ? -1 : 1; });
  const pendientes = eventos.filter(function (x) { return x.siguientePaso && !x.pasoHecho; });
  const idsSemana = {};
  pasaron.forEach(function (x) { idsSemana[x.id] = true; });

  USUARIOS.forEach(function (fila) {
    const nombre = fila[0], admin = fila[1] === true;
    const correo = (CORREOS[nombre] || '').trim();
    if (!correo || fila[1] === 'mama') return;
    if (!pasaron.length && !vienen.length && !pendientes.length && !mensajes.length) return;

    let h = '<div style="font-family:Arial,sans-serif;font-size:15px;line-height:1.5;color:#1d2b33;max-width:600px">';
    h += '<h2 style="color:#2f6f6a;margin:0 0 4px">Resumen de la semana</h2><p style="color:#5b6b72;margin:0 0 16px">Hola, ' + esc(nombre) + '. Esto pasó con Marisol en los últimos 7 días.</p>';

    h += '<h3 style="margin:16px 0 6px">Lo que pasó</h3>';
    h += pasaron.length ? '<ul style="padding-left:18px;margin:0">' + pasaron.map(function (x) {
      let l = '<li style="margin-bottom:8px"><b>' + esc(tipos[x.tipo] || x.tipo) + ': ' + esc(x.titulo) + '</b> — ' + esc(cuando(x.inicio));
      if (x.novedades) l += '<br>' + esc(x.novedades);
      if (x.sintomas) {
        try {
          const sx = JSON.parse(x.sintomas), partes = [];
          ['nauseas', 'cansancio', 'dolor'].forEach(function (k) {
            if (sx[k] !== undefined && sx[k] !== '') partes.push({ nauseas: 'Náuseas', cansancio: 'Cansancio', dolor: 'Dolor' }[k] + ' ' + sx[k] + '/5');
          });
          if (sx.otra) partes.push(sx.otra);
          if (partes.length) l += '<br><span style="color:#5b6b72">Cómo se sintió: ' + esc(partes.join(', ')) + '</span>';
        } catch (z) {}
      }
      return l + '</li>';
    }).join('') + '</ul>' : '<p style="margin:0;color:#5b6b72">No hubo sesiones esta semana.</p>';

    h += '<h3 style="margin:16px 0 6px">Lo que viene (próximas 2 semanas)</h3>';
    h += vienen.length ? '<ul style="padding-left:18px;margin:0">' + vienen.map(function (x) {
      const ac = x.acompanan ? x.acompanan.split('|').join(', ') : '';
      return '<li style="margin-bottom:8px"><b>' + esc(x.titulo) + '</b> — ' + esc(cuando(x.inicio)) + ' (hora Ecuador)<br>' +
        (ac ? 'La acompaña: ' + esc(ac) : '<span style="color:#b5652b"><b>Nadie se ha anotado para acompañarla.</b></span>') + '</li>';
    }).join('') + '</ul>' : '<p style="margin:0;color:#5b6b72">No hay nada agendado.</p>';

    if (pendientes.length) {
      h += '<h3 style="margin:16px 0 6px">Pendientes</h3><ul style="padding-left:18px;margin:0">' +
        pendientes.map(function (x) { return '<li>' + esc(x.siguientePaso) + ' <span style="color:#5b6b72">(de «' + esc(x.titulo) + '»)</span></li>'; }).join('') + '</ul>';
    }

    const misPagos = pagos.filter(function (x) { return idsSemana[x.eventoId] && (admin || x.quienPago === nombre); });
    if (misPagos.length) {
      const total = misPagos.reduce(function (a, x) { return a + (Number(x.monto) || 0); }, 0);
      h += '<h3 style="margin:16px 0 6px">' + (admin ? 'Pagado esta semana' : 'Tu aporte esta semana') + '</h3><p style="margin:0"><b>' + dinero(total) + '</b>' +
        (admin ? ' en ' + misPagos.length + (misPagos.length === 1 ? ' pago' : ' pagos') : '') + '</p>';
    }

    if (mensajes.length) {
      h += '<p style="margin:16px 0 0">💬 Esta semana la familia le dejó ' + mensajes.length + (mensajes.length === 1 ? ' mensaje' : ' mensajes') + ' a Marisol.</p>';
    }

    h += '<p style="margin:24px 0 0"><a href="' + DIRECCION_APP + '" style="color:#2f6f6a;font-weight:bold">Abrir el Cuaderno familia</a> (usa tu propio enlace)</p></div>';

    MailApp.sendEmail({ to: correo, subject: 'Resumen de la semana — Cuaderno familia', htmlBody: h });
  });
}
